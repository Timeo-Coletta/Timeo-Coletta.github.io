# clean world bank data
clean_wb_data <- function(df, value_name) {
  df |>
    select(`Country Name`, `Country Code`, matches("^[0-9]{4}$")) |>
    pivot_longer(
      cols = matches("^[0-9]{4}$"),
      names_to = "year",
      values_to = value_name
    ) |>
    mutate(year = as.numeric(year)) |>
    filter(year >= 2000 & year <= 2020) |>
    rename(Country.Name = `Country Name`,
           Country.Code = `Country Code`)
}

# apply cleaning function to each world bank dataset
co2_clean <- clean_wb_data(co2_raw, "co2_per_capita")

trade_clean <- clean_wb_data(trade_raw, "trade_pct_gdp")

# clean OECD dataset
oecd_clean <- oecd_raw |>
  rename(Country.Name = `Time period...1`) |>
  pivot_longer(cols = matches("^[0-9]{4}$"), names_to = "year", values_to = "eps_index") |>
  fill(Country.Name, .direction = "down") |>
  drop_na(eps_index) |>
  mutate(
    year = as.numeric(year),
    eps_index = as.numeric(eps_index),
    Country.Name = str_replace_all(Country.Name, "[·\\u2007\\u202F]", "") |>
      str_squish() |>
      str_replace_all("T[üu]rkey|T[üu]rkiye", "Turkiye")
  ) |>
  mutate(
    Country.Name = case_when(
      str_detect(Country.Name, regex("Brazil", ignore_case = TRUE)) ~ "Brazil",
      str_detect(Country.Name, regex("People’s Republic of", ignore_case = TRUE)) ~ "China",
      str_detect(Country.Name, regex("Russia", ignore_case = TRUE)) ~ "Russian Federation",
      str_detect(Country.Name, regex("South Africa", ignore_case = TRUE)) ~ "South Africa",
      str_detect(Country.Name, regex("Indonesia", ignore_case = TRUE)) ~ "Indonesia",
      str_detect(Country.Name, regex("India", ignore_case = TRUE)) ~ "India",
      Country.Name == "Korea" ~ "Korea, Rep.",
      TRUE ~ Country.Name)
  ) |>
  filter(year >= 2000 & year <= 2020) |>
  select(Country.Name, year, eps_index)

# extract coefficients for J-curve visualization
h1_coefs <- coef(summary(model_h1))
eps_vars <- c("eps_index", "eps_lag1", "eps_lag2", "eps_lag3")
eps_coefs_h1 <- h1_coefs[rownames(h1_coefs) %in% eps_vars, , drop = FALSE]
eps_df_h1 <- data.frame(
  lag = 0:3,
  coefficient = eps_coefs_h1[, "Estimate"],
  se = eps_coefs_h1[, "Std. Error"]
) |>
  mutate(
    ci_lower = coefficient - 1.96 * se,
    ci_upper = coefficient + 1.96 * se,
    cumulative_effect = cumsum(coefficient)
  )

# test for J-curve: early negative, later positive
j_curve_test_h1 <- list(
  early_effect = eps_df_h1$coefficient[1],
  late_effect = eps_df_h1$cumulative_effect[4]
)
